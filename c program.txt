#include<stdio.h>
int main()
{ 
    int no;
    printf("Enter a number:");
    scanf("%d",&no);
    if(no>=0 && no<10)
    printf("Single Digit");
     else if(no>=10 && no<100)
    printf("Two Digit");
     else if(no>=100 && no<1000)
    printf("Three Digit");
     else if(no>=1000 && no<10000)
    printf("Four Digit");
     else if(no>=10000 && no<100000)
    printf("Five Digit");
    else
    printf("Not Declared");
    return 0;
}
